% One BLADE outer step: inner retain repair -> forget step -> dual update of lambda.
% The forget step and the lambda update together form the ALM constraint (retain loss <= eps).
\begin{tikzpicture}[
    stage/.style 2 args={draw=#1, fill=#1!8, line width=3pt, rounded corners=12pt,
      text width=#2, minimum height=3.6cm, align=center, inner sep=10pt},
    flow/.style={-{Triangle[length=18pt,width=20pt]}, line width=4pt, color=psu-beaver},
  ]
  \node[stage={psu-beaver}{11cm}] (inner) {%
    {\color{psu-beaver}\psuhighsize\textbf{Repair}}\\[0.8ex]
    $K$ fast steps\\cross-entropy loss};
  \node[stage={psu-pink}{12.5cm}, right=2.8cm of inner] (outer) {%
    {\color{psu-pink}\psuhighsize\textbf{Forget}}\\[0.8ex]
    one step\\clamped entropy loss};
  \node[stage={psu-navy}{10cm}, right=2.4cm of outer] (dual) {%
    {\color{psu-navy}\psuhighsize\textbf{Update $\lambda$}}\\[0.8ex]
    asymmetric ratchet};
  \draw[flow] (inner) -- (outer);
  \draw[flow] (outer) -- (dual);
  \coordinate (low) at ([yshift=-1.7cm]outer.south);
  \draw[flow] (dual.south) -- (dual.south |- low) -| node[pos=0.25, fill=white, inner sep=6pt] {repeat} (inner.south);
  \begin{scope}[on background layer]
    \node[draw=psu-navy, dashed, line width=3pt, rounded corners=14pt,
      fit=(outer)(dual), inner xsep=0.8cm, inner ysep=0.6cm,
      label={[anchor=south, inner sep=10pt, text=psu-navy, font=\bfseries]north:%
        ALM: forget only while retain loss $\le \varepsilon$}] {};
  \end{scope}
\end{tikzpicture}
