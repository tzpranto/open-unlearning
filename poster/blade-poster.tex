        \begin{column}{.485\linewidth}
          \begin{psublock}{Why is LLM unlearning hard?}
            \begin{itemize}
              \item Deployed LLMs need continuous correction: GDPR deletion requests, copyrighted text, hazardous knowledge, misinformation.
              \item Forget and retain knowledge share \textbf{entangled representations}: every step risks collateral damage that compounds.
              \item Fixed-weight forget/retain balancing yields three failure modes:
                \begin{itemize}
                  \item \textbf{Unbounded forgetting:} gradient ascent\autocite{jang2023knowledge} has arbitrarily large gradients, destroying utility.
                  \item \textbf{Static trade-off:} GradDiff, NPO\autocite{zhang2024negative} oscillate without self-correction.
                  \item \textbf{Over-forgetting:} unclamped entropy\autocite{yuan2025closer} keeps pushing tokens that are already forgotten.
                \end{itemize}
            \end{itemize}
          \end{psublock}

          \begin{psublock}{BLADE at a glance}
            \raggedright Unlearning as a \textbf{constrained bilevel problem}: forget as much as possible while the retain loss stays within budget~$\varepsilon$.
            The constraint residual is $r = \Loss_{\text{CE}}(\theta;\mathcal{B}'_{\text{ret}}) - \varepsilon$ on a fresh retain batch, so $r > 0$ means the budget is violated.
            \[
              \min_{\theta}\ \Loss_{\text{fgt}}(\theta^*;\Dforget)
              \quad\text{s.t.}\quad \Loss_{\text{CE}}(\theta^*;\Dretain) \le \varepsilon,
              \qquad \theta^* \approx \argmin_{\theta}\ \Loss_{\text{CE}}(\theta;\Dretain)
            \]
            \vspace{-0.5ex}
            \begin{center}
              \input{figs/tikz-blade-loop}
            \end{center}
          \end{psublock}

          \begin{psublock}{\circnum{1} Repair-first bilevel structure}
            \begin{itemize}
              \item \textbf{Inner loop = retain repair}: $K$ fast SGD steps restore retain damage \emph{before} each forget step.
              \item \textbf{Outer loop = one small forget step}, sized so the next inner loop can fully repair it ($\eta_{\text{in}} > \eta_{\text{out}}$).
              \item \textbf{Retain in the inner loop} (contrary to SCRUB\autocite{kurmanji2023towards}, BLUR\autocite{reisizadeh2026blur}): forgetting in the unconstrained inner loop would give $K$ unopposed destructive steps per outer step.
            \end{itemize}
          \end{psublock}

          \begin{psublock}{\circnum{2} Augmented Lagrangian with asymmetric dual update}
            \begin{columns}[c]
              \begin{column}{0.5\textwidth}
                \[
                  \Loss_{\text{outer}} = \Loss_{\text{fgt}} + \lambda r + \tfrac{\rho}{2}\big[\max(0,r)\big]^2
                \]
              \end{column}
              \begin{column}{0.46\textwidth}
                \[
                  \lambda \leftarrow
                  \begin{cases}
                    \lambda + \rho\,|r| & r > 0\\
                    \lambda - \alpha\rho\,|r| & r \le 0,\ \alpha \ll 1
                  \end{cases}
                \]
              \end{column}
            \end{columns}
            \begin{itemize}
              \item $r = \Loss_{\text{ret}} - \varepsilon$ is measured on a fresh retain batch, and $\varepsilon$ is set relative to the model's own retain loss (no manual calibration).
              \item An \textbf{inequality} constraint: there is no penalty while retain loss stays below~$\varepsilon$.
              \item \textbf{Ratchet:} $\lambda$ jumps after a violation and decays only slowly. Symmetric updates (PDU\autocite{entesari2026constrained}) oscillate instead.
            \end{itemize}
          \end{psublock}

          \begin{psublock}{\circnum{3} Clamped entropy forget loss}
            \begin{columns}[c]
              \begin{column}{0.53\textwidth}
                \[
                  \Loss_{\text{fgt}} = \frac{1}{N}\sum_{t=1}^{N} \max\!\big(0,\ \tau\log V - H(p_t)\big)
                \]
                \begin{itemize}
                  \item \textbf{Bounded:} $\Loss_{\text{fgt}} \in [0, \tau\log V]$.
                  \item \textbf{Self-stabilising:} the gradient is \emph{exactly zero} once a token is forgotten.
                  \item \textbf{Reference-free:} no frozen model, unlike NPO.
                  \item \textbf{Vocabulary-invariant} $\tau$ (32K or 128K tokens).
                \end{itemize}
              \end{column}
              \begin{column}{0.43\textwidth}
                \centering
                \adjustbox{max width=\linewidth}{\input{figs/tikz-clamped-entropy}}
              \end{column}
            \end{columns}
            \begin{takeaway}
              \raggedright\textbf{LoRA as a structural constraint:} adapters on all projections, $W' = W_0 + \tfrac{\alpha}{r}BA$. The low-rank bottleneck limits weight drift, and disabling the adapters restores the original model.
            \end{takeaway}
          \end{psublock}

          \begin{psublock}{Experimental setup}
            \begin{itemize}
              \item \textbf{TOFU}\autocite{maini2024tofu}: Synthetic authors, Llama-3.2-1B/3B-Instruct, forget 1/5/10\%.
              \item \textbf{MUSE}\autocite{shi2025muse}: Books (Harry Potter) and News (BBC), Llama-2-7B.
              \item \textbf{KnowUndo}\autocite{tian2024knowundo}: Copyright and privacy, LoRA-fine-tuned Llama-2-7B-chat.
              \item \textbf{Baselines} (OpenUnlearning): GradAscent, GradDiff, NPO, SimNPO\autocite{fan2025simplicity}, RMU, BLURNPO, PDU.
              \item \textbf{Metrics:} per-benchmark harmonic mean (HM) of forget and retain axes, plus an LLM-judge composite HM$_\text{J}$ for semantic leakage. All results are averaged over 5 seeds.
            \end{itemize}
          \end{psublock}

          \begin{psufinalblock}{Takeaways and limitations}
            \begin{itemize}
              \item A \textbf{bounded} forget loss, an \textbf{adaptive} retain constraint and a \textbf{low-rank} parameter space combine into a self-correcting unlearner.
              \item Gains hold across benchmarks, \textbf{under scaling} and \textbf{under repeated application}.
              \item \textbf{Smooth training dynamics}, and extremely robust to hyperparameter choice.
              \item \textbf{Adversarially robust} against black-box attacks.
              \item \textbf{Limitations:} Vulnerable to re-learning attacks. Not yet tested on concept-level unlearning or in multimodal settings.
            \end{itemize}
          \end{psufinalblock}

            \begin{findingblock}{Finding 2. Robust under stress, where the strongest baseline collapses.}{navy}
              \begin{columns}[c]
                \begin{column}{0.55\textwidth}
                  \begin{itemize}
                    \item MUSE News, head-to-head against PDU, the strongest News baseline.
                    \item \textbf{Scalability:} forget set grown up to $4\times$. PDU collapses at $4\times$ (HM = 0.005).
                    \item \textbf{Sustainability:} sequential unlearning requests. PDU drops from 0.58 to 0.02 after only 2 steps.
                    \item \method stays within \textbf{HM 0.523--0.552} under every condition.
                  \end{itemize}
                  \begin{callout}
                    Bounded loss, the LoRA bottleneck and the ALM ratchet together prevent the model collapse at scale and under repeated application.
                  \end{callout}
                \end{column}
                \begin{column}{0.43\textwidth}
                  \centering
                  \includegraphics[width=0.92\linewidth]{figures/fig_scalability_sustainability}
                \end{column}
              \end{columns}
            \end{findingblock}

            \begin{findingblock}{Finding 3. Smooth, self-correcting three-phase dynamics.}{beaver}
              \begin{columns}[T]
                \begin{column}{0.5\textwidth}
                  \centering
                  \includegraphics[width=\linewidth]{figures/fig_muse_books_dynamics}\\
                  {\small\method: warm-up $\to$ spike-and-ratchet $\to$ convergence}
                \end{column}
                \begin{column}{0.5\textwidth}
                  \centering
                  \includegraphics[width=\linewidth]{figures/fig_muse_books_PDU_dynamics}\\
                  {\small PDU: recurring retain oscillations}
                \end{column}
              \end{columns}
              \begin{enumerate}
                \item \textbf{Warm-up:} the forget loss falls while retain stays within $\varepsilon$.
                \item \textbf{Spike-and-ratchet:} a transient retain violation permanently raises $\lambda$.
                \item \textbf{Smooth convergence:} both losses decrease monotonically under the elevated $\lambda$.
              \end{enumerate}
              The same pattern appears on every benchmark, model scale and dataset type, which makes the forget--retain trade-off interpretable and controllable.
            \end{findingblock}

            \begin{findingblock}{Finding 4. BLADE's components complement each other.}{navy}
              \begin{columns}[T]
                \begin{column}{0.5\textwidth}
                  \centering\small
                  \setlength{\tabcolsep}{7pt}
                  \begin{tabular}{@{}lccc@{}}
                    \toprule
                    MUSE Books & rk\up & HM\up & HM$_\text{J}$\up \\
                    \midrule
                    Full \method & .658 & \best{.823} & \best{.814} \\
                    \midrule
                    ALM off & .092 & .233 & .117 \\
                    No LoRA (full FT) & .306 & .459 & .550 \\
                    Swapped bilevel & .651 & .506 & .655 \\
                    Unclamped ($\tau{=}1$) & .604 & .780 & .785 \\
                    \midrule
                    GA forget loss & .060 & .160 & .093 \\
                    NPO forget loss & .662 & .009 & .495 \\
                    Logit-margin loss & .000 & .000 & .000 \\
                    \bottomrule
                  \end{tabular}
                \end{column}
                \begin{column}{0.48\textwidth}
                  \begin{itemize}
                    \item \textbf{ALM} is the most critical piece: without it, retain collapses.
                    \item \textbf{LoRA} prevents representational drift (full fine-tuning: rk .658 $\to$ .306).
                    \item \textbf{Swapping the loops} makes $\lambda$ diverge, which confirms the repair-first principle.
                    \item Only a \textbf{bounded, self-saturating} forget loss is compatible with the constrained bilevel framework.
                  \end{itemize}
                \end{column}
              \end{columns}
            \end{findingblock}

            \begin{findingblock}{Finding 5. Resilient to black-box attacks and memory-efficient.}{beaver}
              \begin{itemize}
                \item \textbf{Extraction}, using paraphrased or perturbed queries: best adversarial HM on TOFU 1B on every split.
                \item \textbf{Jailbreak:} lowest attack success rate on every split (0.025 vs.\ PDU 0.087 on fgt01).
                \item \textbf{Membership inference:} the only method whose PrivLeak score is positive on every split (no membership leakage).
                \item \textbf{Compute:} $2$--$5\times$ lower peak GPU memory than every baseline, for $1.5$--$2\times$ the wall-clock time.
              \end{itemize}
            \end{findingblock}
